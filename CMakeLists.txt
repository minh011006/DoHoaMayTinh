cmake_minimum_required(VERSION 3.16)
project(DoHoaMayTinh LANGUAGES CXX)

add_library(vase_mesh STATIC src/vase_mesh.cpp)
target_include_directories(vase_mesh PUBLIC include)
target_compile_features(vase_mesh PUBLIC cxx_std_17)

include(CTest)
if(BUILD_TESTING)
    add_executable(vase_mesh_check tests/vase_mesh_check.cpp)
    target_link_libraries(vase_mesh_check PRIVATE vase_mesh)
    add_test(NAME vase_mesh_check COMMAND vase_mesh_check)
endif()

add_library(bezier_profile STATIC src/bezier_profile.cpp)
target_include_directories(bezier_profile PUBLIC include)
target_compile_features(bezier_profile PUBLIC cxx_std_17)

option(BUILD_BEZIER_EDITOR "Build the FreeGLUT profile editor" ON)
if(BUILD_BEZIER_EDITOR)
    find_package(OpenGL REQUIRED)
    find_package(FreeGLUT CONFIG QUIET)
    if(TARGET FreeGLUT::freeglut)
        set(EDITOR_GLUT_TARGET FreeGLUT::freeglut)
    else()
        find_package(GLUT REQUIRED)
        set(EDITOR_GLUT_TARGET GLUT::GLUT)
    endif()
    add_executable(bezier_editor src/bezier_editor.cpp)
    target_link_libraries(bezier_editor PRIVATE bezier_profile OpenGL::GL ${EDITOR_GLUT_TARGET})
endif()
if(BUILD_TESTING)
    add_executable(bezier_profile_check tests/bezier_profile_check.cpp)
    target_link_libraries(bezier_profile_check PRIVATE bezier_profile vase_mesh)
    add_test(NAME bezier_profile_check COMMAND bezier_profile_check)
endif()
